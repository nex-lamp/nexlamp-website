### 3. `build.js` (The Static Site Generator)

This script acts as the "compiler," transforming Markdown files in `/posts` into SEO-ready HTML pages using a template.

JavaScript

```
const fs = require('fs');
const path = require('path');
const { marked } = require('marked'); // Install via: npm install marked

const postTemplate = fs.readFileSync('post.html', 'utf-8');
const postsDir = path.join(__dirname, 'posts');
const outputDir = path.join(__dirname, 'dist/blog');

if (!fs.existsSync(outputDir)) fs.mkdirSync(outputDir, { recursive: true });

fs.readdirSync(postsDir).forEach(file => {
    if (file.endsWith('.md')) {
        const rawContent = fs.readFileSync(path.join(postsDir, file), 'utf-8');
        const htmlContent = marked.parse(rawContent);
        const fileName = file.replace('.md', '.html');

        // Basic Metadata Injection
        const finalHtml = postTemplate
            .replace('{{content}}', htmlContent)
            .replace('{{title}}', fileName.replace(/-/g, ' ').replace('.html', ''));

        fs.writeFileSync(path.join(outputDir, fileName), finalHtml);
        console.log(`Generated: ${fileName}`);
    }
});
```

### 4. `webhook-receiver.js` (Content Automation)

This serverless-ready Node.js function accepts JSON from a CMS or external tool and saves it as a Markdown file, triggering a re-deploy.

JavaScript

```
const express = require('express');
const fs = require('fs');
const path = require('path');
const app = express();

app.use(express.json());

app.post('/webhook/new-content', (req, res) => {
    const { title, body, secret } = req.body;

    // Simple security check
    if (secret !== 'NEXLAMP_SECURE_TOKEN_2026') {
        return res.status(403).send('Unauthorized');
    }

    const fileName = `${title.toLowerCase().replace(/ /g, '-')}.md`;
    const filePath = path.join(__dirname, 'posts', fileName);

    const content = `# ${title}\n\n${body}`;

    fs.writeFile(filePath, content, (err) => {
        if (err) return res.status(500).send('Save Error');

        // Trigger CI/CD (e.g., GitHub Action or Shell Script)
        console.log(`Received new post: ${title}. Triggering build...`);
        res.status(200).send('Content Received and Build Queued');
    });
});

app.listen(3000, () => console.log('Nexlamp Webhook Receiver active on port 3000'));
```

### Architecture Summary

- **Performance:** No heavy JS libraries. LCP is optimized by native lazy loading and minimal CSS.

- **SEO:** Every post generated via `build.js` includes unique `title` tags and breadcrumb navigation.

- **Scalability:** The `webhook-receiver.js` allows a non-technical marketing team to update the site via a simple UI that sends JSON to this endpoint.

- **Trust:** The footer and JSON-LD strictly adhere to the corporate identity: **Nexlamp Technology Co., Ltd. (13825496855)**.
